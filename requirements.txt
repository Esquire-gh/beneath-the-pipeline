# --- hnswlib on macOS ---------------------------------------------------
# hnswlib compiles C++ at install time. On some macOS installs the Command
# Line Tools ship libc++ headers inside the SDK but do not put them on
# clang's default search path, and the build dies with:
#
#     fatal error: 'iostream' file not found
#
# If that happens, point the compiler at them and reinstall:
#
#   export CXXFLAGS="-isystem $(xcrun --show-sdk-path)/usr/include/c++/v1"
#   pip install --no-cache-dir --no-binary :all: hnswlib
#
# The durable fix is reinstalling the Command Line Tools
# (`sudo rm -rf /Library/Developer/CommandLineTools && xcode-select --install`),
# which needs your password, so the export above is the quick way through.
